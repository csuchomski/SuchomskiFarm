  setAssetKindAsset = () => this.setState(s=>({assetForm:{...s.assetForm, kind:'asset'}}));
  setAssetKindLiability = () => this.setState(s=>({assetForm:{...s.assetForm, kind:'liability'}}));
  handleAssetBizChange = (e) => { const v = e.target.value; const id = /^\d+$/.test(v) ? Number(v) : v; this.setState(s=>({assetForm:{...s.assetForm, businessId:id}})); };
  handleAssetName = (e) => this.setState(s=>({assetForm:{...s.assetForm, name:e.target.value}}));
  handleAssetValue = (e) => this.setState(s=>({assetForm:{...s.assetForm, value:e.target.value}}));
  saveAsset = async () => {
    const f = this.state.assetForm;
    if (!f.businessId || !f.name.trim() || f.value === '') return;
    try {
      const rows = await this.rest('/ledger_assets', { method:'POST', body:{ business_id:f.businessId, kind:f.kind, name:f.name.trim(), value:Number(f.value) } });
      const a = rows[0];
      const rec = { id:a.id, businessId:a.business_id, kind:a.kind, name:a.name, value:Number(a.value) };
      this.setState(s=>({assets:[...s.assets, rec], showAssetModal:false}));
    } catch (e) { this.setState({authError:e.message}); }
  };
  deleteAsset = async (id) => {
    if (!window.confirm('Delete this entry?')) return;
    try {
      await this.rest('/ledger_assets?id=eq.'+id, { method:'DELETE' });
      this.setState(s=>({assets:s.assets.filter(a=>a.id!==id)}));
    } catch (e) { this.setState({authError:e.message}); }
  };

    const kpiIncome = sum(scopeTxnsPeriod,'income');
    const kpiExpense = sum(scopeTxnsPeriod,'expense');
    const kpiNet = kpiIncome - kpiExpense;
    const periodEndDate = this.periodEnd();
    const kpiCash = scopeIds.reduce((tot,bid)=> tot + this.computedAccounts(bid).reduce((sum,a)=> sum + ((a.businessId===bid)?a.openingBalance:0) + this.acctBalance(bid, a.name, periodEndDate), 0), 0);

      return {label:MONTHS[i], income:sum(txns,'income'), expense:sum(txns,'expense')};

    const monthlyChart = monthly.map(m=>({label:m.label, incomeH: Math.round(m.income/maxVal*100), expenseH: Math.round(m.expense/maxVal*100)}));

    const expenseCatTotals = {};
    scopeTxnsPeriod.filter(t=>t.type==='expense').forEach(t=>{ expenseCatTotals[t.category]=(expenseCatTotals[t.category]||0)+t.amount; });
    const topExpenseArr = Object.entries(expenseCatTotals).sort((a,b)=>b[1]-a[1]).slice(0,5);
    const maxExpense = Math.max(1, ...topExpenseArr.map(e=>e[1]));
    const topExpenseCategories = topExpenseArr.map(([category,amount])=>({category, amountLabel:fmtMoney(amount), pct: Math.round(amount/maxExpense*100)}));

    const recentTxns = [...scopeTxnsPeriod].sort((a,b)=>b.date.localeCompare(a.date)).slice(0,6).map(t=>({
      category:t.category, dateLabel:fmtDateLabel(t.date), bizName:this.bizName(t.businessId), showBiz:false,
      amountLabel:(t.type==='income'?'+':'-')+fmtMoney(t.amount), amountColor: t.type==='income'?ACCENT:NEG
    }));

    const filteredTxnsList = s.transactions.filter(t=>
      scopeIds.includes(t.businessId) && this.withinPeriod(t.date) &&
      (s.filterType==='all'||t.type===s.filterType) &&
      (s.filterCategory==='all'||t.category===s.filterCategory) &&
      (s.search.trim()==='' || (t.category+' '+(t.note||'')+' '+(t.payer||'')+' '+(t.account||'')+' '+this.bizName(t.businessId)).toLowerCase().includes(s.search.toLowerCase()))
    ).sort((a,b)=>b.date.localeCompare(a.date)).map(t=>({
      id:t.id, dateLabel:fmtDateLabel(t.date), bizName:this.bizName(t.businessId), showBiz:false,
      category:t.category, note:t.note, subLabel:[t.payer, t.account, t.note].filter(Boolean).join(' \u00b7 '), typeLabel:t.type==='income'?'Income':'Expense',
      typeBg: t.type==='income'?ACCENT_SOFT:NEG_SOFT, typeColor: t.type==='income'?'oklch(0.32 0.06 150)':'oklch(0.36 0.08 25)',
      amountLabel:(t.type==='income'?'+':'-')+fmtMoney(t.amount), amountColor: t.type==='income'?ACCENT:NEG,

    }));

      const expenseRows = def.expense.map(c=>({category:c, amount:sum(txns,'expense',c)})).filter(r=>r.amount>0).map(r=>({...r, amountLabel:fmtMoney(r.amount)}));

      const cashRows = this.computedAccounts(b.id).map(a=>{
        const value = ((a.businessId===b.id)?a.openingBalance:0) + this.acctBalance(b.id, a.name, periodEndDate);
        return {name:a.name, value, valueLabel:fmtMoney(value)};
      }).filter(r=>r.value!==0 || r.name.toLowerCase()==='cash');
      const cashTotal = cashRows.reduce((a,x)=>a+x.value,0);
      const assetRows = rows.filter(r=>r.kind==='asset' && !/cash/i.test(r.name) && !acctNamesLower.has(r.name.toLowerCase())).map(r=>({...r, valueLabel:fmtMoney(r.value), onDelete:()=>this.deleteAsset(r.id)}));
      const assetTotal = cashTotal + assetRows.reduce((a,x)=>a+x.value,0);
      const liabilityRows = rows.filter(r=>r.kind==='liability').map(r=>({...r, valueLabel:fmtMoney(r.value), onDelete:()=>this.deleteAsset(r.id)}));
      const liabilityTotal = liabilityRows.reduce((a,x)=>a+x.value,0);
      const equity = assetTotal - liabilityTotal;
      return {id:b.id, name:b.name, scheduleShort:def.scheduleShort, cashRows, cashTotalLabel:fmtMoney(cashTotal), assetRows, assetTotalLabel:fmtMoney(assetTotal), liabilityRows, liabilityTotalLabel:fmtMoney(liabilityTotal), equityLabel:fmtMoney(equity)};

        const txns = s.transactions.filter(t=>t.businessId===b.id && Number(t.date.slice(0,4))===s.period.year && Number(t.date.slice(5,7))-1===mi);

        running += net;
        return {label:MONTHS[mi]+' '+s.period.year, incomeLabel:fmtMoney(inc), expenseLabel:fmtMoney(exp), netLabel:(net>=0?'+':'')+fmtMoney(net), netColor: net>=0?ACCENT:NEG, runningLabel:fmtMoney(running), _net:net};
      });
      const totalNet = rows.reduce((a,x)=>a+x._net,0);
      return {id:b.id, name:b.name, scheduleShort:def.scheduleShort, startCashLabel:fmtMoney(startCash), rows, totalNetLabel:(totalNet>=0?'+':'')+fmtMoney(totalNet), endCashLabel:fmtMoney(running)};
    });

      const txns = s.transactions.filter(t=>t.businessId===b.id && this.withinPeriod(t.date));
      const def = BIZ_TYPES[b.type];
      const incomeRows = def.income.map(c=>({category:c, amountLabel:fmtMoney(sum(txns,'income',c))}));
      const expenseRows = def.expense.map(c=>({category:c, amountLabel:fmtMoney(sum(txns,'expense',c))}));
      const incomeTotal = sum(txns,'income'), expenseTotal = sum(txns,'expense'), net = incomeTotal-expenseTotal;
      return {id:b.id, name:b.name, scheduleFull:def.scheduleFull, incomeRows, incomeTotalLabel:fmtMoney(incomeTotal), expenseRows, expenseTotalLabel:fmtMoney(expenseTotal), netLine:'Net profit / loss ('+def.scheduleShort+')', netLabel:fmtMoney(net), netColor: net>=0?ACCENT:NEG};
    });

    // txn form category options
    const accountOptions = ['Cash', ...s.accounts.map(a=>a.name).filter(x=>x.toLowerCase()!=='cash')];
    const acctDraftList = s.acctDrafts.map(d=>({
      id:d.id, name:d.name, businessId:d.businessId, opening:d.opening,
      onName:(e)=>this.editDraft(d.id,{name:e.target.value}),
      onBiz:(e)=>{const v=e.target.value; this.editDraft(d.id,{businessId: v===''?'':(/^\d+$/.test(v)?Number(v):v)});},
      onOpening:(e)=>this.editDraft(d.id,{opening:e.target.value}),
      onDelete:()=>this.deleteAccount(d.id)
    }));
    const txnBiz = this.bizById(s.txnForm.businessId);
    const txnFormCategoryOptions = txnBiz ? BIZ_TYPES[txnBiz.type][s.txnForm.type] : [];

    return {
      isAuthReady: true, isAuthLoading:false, isAuthDenied:false, authError:'',
      navDashboardStyle: this.navStyle(s.view==='dashboard'),
      navTransactionsStyle: this.navStyle(s.view==='transactions'),
      navReportsStyle: this.navStyle(s.view==='reports'),
      goDashboard: this.goDashboard, goTransactions: this.goTransactions, goReports: this.goReports,
      businessList, openAddBusiness: this.openAddBusiness,

      navAccountsStyle: this.navStyle(s.view==='accounts'), goAccounts: this.goAccounts, acctSaved: s.acctSaved && !s.acctError,

      modeYearStyle: this.pillStyle(s.period.mode==='year'), modeQuarterStyle: this.pillStyle(s.period.mode==='quarter'),
      setModeYear: this.setModeYear, setModeQuarter: this.setModeQuarter,
      isQuarterMode: s.period.mode==='quarter', quarterOptions, period: s.period,
      handleQuarterChange: this.handleQuarterChange, handleYearChange: this.handleYearChange,
      periodYearOptions, periodLabel, openAddTxn: this.openAddTxn,

      kpiIncomeLabel: fmtMoney(kpiIncome), kpiExpenseLabel: fmtMoney(kpiExpense),
      kpiNetLabel: fmtMoney(kpiNet), kpiNetColor: kpiNet>=0?ACCENT:NEG, kpiCashLabel: fmtMoney(kpiCash),
      monthlyChart,
      hasTopExpenses: topExpenseCategories.length>0, noTopExpenses: topExpenseCategories.length===0, topExpenseCategories,

      filterCategory: s.filterCategory, handleFilterCategoryChange: this.handleFilterCategoryChange,
      categoryFilterOptions, showBizColumn: false, txnGridExtra: '',
      hasFilteredTxns: filteredTxnsList.length>0, noFilteredTxns: filteredTxnsList.length===0, filteredTxnsList,

      reportsTab: s.reportsTab,

  }
